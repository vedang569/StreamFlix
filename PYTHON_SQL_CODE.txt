import os
import mysql.connector
from mysql.connector import pooling
from dotenv import load_dotenv

load_dotenv()

# Database Connection Settings
db_config = {
    "host":     os.getenv("DB_HOST", "localhost"),
    "port":     int(os.getenv("DB_PORT", 3306)),
    "user":     os.getenv("DB_USER", "root"),
    "password": os.getenv("DB_PASSWORD", ""),
    "database": os.getenv("DB_NAME", "streamflix"),
}

# Thread-safe MySQL Connection Pool (Pool size: 5)
pool = pooling.MySQLConnectionPool(
    pool_name="streamflix_pool",
    pool_size=5,
    **db_config
)

def get_db():
    """Returns an active MySQL connection from the pool for current request context (g)."""
    if "db" not in g:
        g.db = pool.get_connection()
    return g.db

@app.teardown_appcontext
def close_db(exc):
    """Automatically closes and releases connection back to pool after request."""
    db = g.pop("db", None)
    if db is not None:
        db.close()

def query(sql, params=None, fetchone=False, skip_log=False):
    """
    Executes a SELECT query and returns dictionary row results.
    Prevents SQL injection using parameterized tuple binding (%s).
    """
    start_time = time.time()
    conn = get_db()
    cur = conn.cursor(dictionary=True)
    cur.execute(sql, params or ())
    rows = cur.fetchone() if fetchone else cur.fetchall()
    cur.close()
    duration_ms = (time.time() - start_time) * 1000
    if not skip_log:
        log_sql_execution(sql, params, duration_ms, is_write=False)
    return rows


def execute(sql, params=None, skip_log=False):
    """
    Executes INSERT / UPDATE / DELETE state-changing statements and commits.
    Returns last inserted auto-increment ID.
    """
    start_time = time.time()
    conn = get_db()
    cur = conn.cursor()
    cur.execute(sql, params or ())
    conn.commit()
    last_id = cur.lastrowid
    cur.close()
    duration_ms = (time.time() - start_time) * 1000
    if not skip_log:
        log_sql_execution(sql, params, duration_ms, is_write=True)
    return last_id

3. REAL-TIME SQL EXECUTION LOGGING ENGINE (from app.py)

from collections import deque
from datetime import datetime

# Ring buffer storing last 100 executed database statements
query_log_buffer = deque(maxlen=100)

def log_sql_execution(sql, params, duration_ms, is_write=False):
    """Logs SQL statement execution time, params, and operation type for DB Inspector UI."""
    clean_sql = " ".join(sql.strip().split())
    if not clean_sql:
        return
    first_word = clean_sql.split()[0].upper()

    formatted_params = None
    if params:
        if isinstance(params, (tuple, list)):
            formatted_params = [str(p) for p in params]
        elif isinstance(params, dict):
            formatted_params = {k: str(v) for k, v in params.items()}
        else:
            formatted_params = str(params)

    entry = {
        "id": len(query_log_buffer) + 1,
        "timestamp": datetime.now().strftime("%H:%M:%S.%f")[:-3],
        "sql": clean_sql,
        "type": first_word,
        "params": formatted_params,
        "duration_ms": round(duration_ms, 2),
        "is_write": is_write
    }
    query_log_buffer.appendleft(entry)

    tables_info = []
    for tbl in allowed_tables:

        tables_info.append({"name": tbl, "rows": res["count"] if res else 0})
    return jsonify(tables_info)

import os
from dotenv import load_dotenv
import mysql.connector

load_dotenv()

def init_db():
    host = os.getenv("DB_HOST", "localhost")
    port = int(os.getenv("DB_PORT", 3306))
    user = os.getenv("DB_USER", "root")
    password = os.getenv("DB_PASSWORD", "")

    print(f"Connecting to MySQL server at {host}:{port} as '{user}'...")
    conn = mysql.connector.connect(host=host, port=port, user=user, password=password)
    cursor = conn.cursor()

    with open("schema.sql", "r", encoding="utf-8") as f:
        sql_script = f.read()

    print("Executing database schema & seed data...")
    for result in cursor.execute(sql_script, multi=True):
        if result.with_rows:
            result.fetchall()

    conn.commit()
    cursor.close()
    conn.close()
    print("\n[OK] Database 'streamflix' successfully created and seeded!")
